# This directory can be tested independently of Kokkos and the physics modules.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.10)
  project(multirate_balance_tests LANGUAGES CXX)
  enable_testing()
endif()
get_filename_component(MULTIRATE_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(multirate_balance_tests balance_tests.cpp
  ${MULTIRATE_REPO_ROOT}/src/mesh/multirate_load_balance.cpp)
target_include_directories(multirate_balance_tests PRIVATE ${MULTIRATE_REPO_ROOT}/src)
target_compile_features(multirate_balance_tests PRIVATE cxx_std_17)
if(TARGET METIS::METIS)
  target_link_libraries(multirate_balance_tests PRIVATE METIS::METIS)
  target_compile_definitions(multirate_balance_tests PRIVATE ATHENA_MULTIRATE_METIS=1)
elseif(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  find_path(MULTIRATE_METIS_INCLUDE_DIR metis.h)
  find_library(MULTIRATE_METIS_LIBRARY NAMES metis)
  find_library(MULTIRATE_GKLIB_LIBRARY NAMES GKlib)
  if(MULTIRATE_METIS_INCLUDE_DIR AND MULTIRATE_METIS_LIBRARY)
    target_include_directories(multirate_balance_tests PRIVATE ${MULTIRATE_METIS_INCLUDE_DIR})
    target_link_libraries(multirate_balance_tests PRIVATE ${MULTIRATE_METIS_LIBRARY})
    if(MULTIRATE_GKLIB_LIBRARY)
      target_link_libraries(multirate_balance_tests PRIVATE ${MULTIRATE_GKLIB_LIBRARY})
    endif()
    target_compile_definitions(multirate_balance_tests PRIVATE ATHENA_MULTIRATE_METIS=1)
  endif()
endif()
add_test(NAME multirate_balance COMMAND multirate_balance_tests)
