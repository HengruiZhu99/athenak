cmake_minimum_required(VERSION 3.16)
project(athenak_multirate_tests LANGUAGES CXX)
enable_testing()
add_executable(test_multirate test_multirate.cpp ../../src/driver/multirate.cpp)
target_compile_features(test_multirate PRIVATE cxx_std_17)
target_include_directories(test_multirate PRIVATE ../../src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(test_multirate PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME multirate_numerics COMMAND test_multirate)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME multirate_exact_order
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/check_order.py
          --executable $<TARGET_FILE:test_multirate>)

option(MULTIRATE_TRANSPORT_TESTS "Build sparse record transport tests" OFF)
if(TARGET Kokkos::kokkos OR MULTIRATE_TRANSPORT_TESTS)
  if(NOT TARGET Kokkos::kokkos)
    find_package(Kokkos CONFIG REQUIRED)
  endif()
  if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    option(MULTIRATE_TEST_MPI "Use MPI in standalone transport tests" OFF)
    set(Athena_ENABLE_MPI ${MULTIRATE_TEST_MPI})
    if(MULTIRATE_TEST_MPI)
      find_package(MPI REQUIRED COMPONENTS CXX)
      set(MULTIRATE_MPI_MACRO 1)
    else()
      set(MULTIRATE_MPI_MACRO 0)
    endif()
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/config.hpp
      "#define SINGLE_PRECISION_ENABLED 0\n#define MPI_PARALLEL_ENABLED ${MULTIRATE_MPI_MACRO}\n")
  endif()
  add_executable(test_multirate_exchange exchange_tests.cpp ../../src/bvals/multirate_exchange.cpp)
  target_compile_features(test_multirate_exchange PRIVATE cxx_std_17)
  target_include_directories(test_multirate_exchange PRIVATE ../../src
    ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_BINARY_DIR})
  target_link_libraries(test_multirate_exchange PRIVATE Kokkos::kokkos)
  if(Athena_ENABLE_MPI)
    target_link_libraries(test_multirate_exchange PRIVATE MPI::MPI_CXX)
    foreach(ranks 2 3)
      add_test(NAME multirate_exchange_${ranks} COMMAND ${MPIEXEC_EXECUTABLE}
        ${MPIEXEC_NUMPROC_FLAG} ${ranks} ${MPIEXEC_PREFLAGS}
        $<TARGET_FILE:test_multirate_exchange> ${MPIEXEC_POSTFLAGS})
    endforeach()
  endif()
  add_test(NAME multirate_exchange COMMAND test_multirate_exchange)
endif()
