cmake_minimum_required(VERSION 3.16)
project(pcgh_intrinsic_oracles LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Kokkos CONFIG REQUIRED)
add_executable(intrinsic_map intrinsic_map.cpp)
target_link_libraries(intrinsic_map PRIVATE Kokkos::kokkos)
target_include_directories(intrinsic_map PRIVATE ../../../src "${ATHENA_CONFIG_DIR}" "${ATHENA_SOURCE_DIR}")
add_executable(intrinsic_rhs intrinsic_rhs.cpp)
target_link_libraries(intrinsic_rhs PRIVATE Kokkos::kokkos)
target_include_directories(intrinsic_rhs PRIVATE ../../../src)
# Diagnostic controls retained for the unresolved CUDA combined-harness failure.
foreach(name intrinsic_probe intrinsic_debug intrinsic_hostinspect intrinsic_probe128 intrinsic_indexed intrinsic_loadorder)
  add_executable(${name} EXCLUDE_FROM_ALL ${name}.cpp)
  target_link_libraries(${name} PRIVATE Kokkos::kokkos)
  target_include_directories(${name} PRIVATE ../../../src)
endforeach()
add_executable(intrinsic_stencil intrinsic_stencil.cpp)
target_link_libraries(intrinsic_stencil PRIVATE Kokkos::kokkos)
target_include_directories(intrinsic_stencil PRIVATE ../../../src "${ATHENA_CONFIG_DIR}" "${ATHENA_SOURCE_DIR}")
